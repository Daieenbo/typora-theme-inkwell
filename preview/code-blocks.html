// 计算斐波那契数列 (JetBrains New Dark)
import { useMemo } from "react";
export function fibonacci(n) {
  const cache = useMemo(() => n <= 1 ? n : 0, [n]);
  if (n <= 1) return n;
  return fibonacci(n - 1) + fibonacci(n - 2);
}
